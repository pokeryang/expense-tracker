云同步设置（Supabase + 密钥门；换设备/清缓存需重新输入密钥）

数据存 Supabase 的 expenses 表（与 daily-records 的 records 表隔离）。为了防止有人从公开的前端源码扒到 Supabase URL+key 就能读写你的账单，本页在数据库侧开了 RLS + 共享密钥：必须请求头带正确密钥才能读写，否则一律拒绝。首次使用请把下面这段 SQL 完整粘到 Supabase 控制台 SQL Editor 执行一次（含密钥），然后在弹窗里输入同一密钥即可解锁；密钥只存本机浏览器 localStorage，不会上传到 GitHub。

-- 1) 建表
create table if not exists public.expenses (
  id text primary key,
  ts date,
  amount numeric,
  category text,
  data jsonb,
  updated_at timestamptz default now()
);

-- 2) 密钥校验函数（RLS policy + verify RPC 都靠它）
create or replace function public.verify_expense_secret() returns boolean
  language sql stable
  as $$
    select current_setting('request.headers', true)::json->>'x-app-secret'
         = '3IaUwV1fOzZCpwKdERumRKfWp70wYU_HxdqCC5lq-y4'
  $$;
grant execute on function public.verify_expense_secret() to anon, authenticated;

-- 3) 开 RLS，仅密钥正确时可读写
alter table public.expenses enable row level security;
drop policy if exists "expenses_all" on public.expenses;
create policy "expenses_all" on public.expenses
  for all to anon, authenticated
  using (public.verify_expense_secret())
  with check (public.verify_expense_secret());
数据实时同步到云端（Supabase expenses 表），本机浏览器同时留一份缓存做离线兜底。建议偶尔「导出备份」为 JSON 存档。
